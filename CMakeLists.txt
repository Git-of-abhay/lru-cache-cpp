cmake_minimum_required(VERSION 3.20)
project(lru_cache_cpp VERSION 1.0.0 LANGUAGES CXX)
option(LRU_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(LRU_TSAN "Enable ThreadSanitizer (separate build)" OFF)
if(LRU_SANITIZERS AND LRU_TSAN)
  message(FATAL_ERROR "Use separate builds for ASan and TSan")
endif()
find_package(Threads REQUIRED)
add_library(lru INTERFACE)
target_include_directories(lru INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(lru INTERFACE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(lru INTERFACE -Wall -Wextra -Wpedantic -Werror)
  if(LRU_SANITIZERS)
    target_compile_options(lru INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(lru INTERFACE -fsanitize=address,undefined)
  elseif(LRU_TSAN)
    target_compile_options(lru INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(lru INTERFACE -fsanitize=thread)
  endif()
endif()
add_executable(lru_cli src/main.cpp)
target_link_libraries(lru_cli PRIVATE lru)
add_executable(lru_tests tests/cache_test.cpp)
target_link_libraries(lru_tests PRIVATE lru Threads::Threads)
add_executable(lru_benchmark benchmarks/benchmark.cpp)
target_link_libraries(lru_benchmark PRIVATE lru)
enable_testing()
add_test(NAME cache_correctness COMMAND lru_tests)
add_test(NAME cli_demo COMMAND lru_cli --demo)
set_tests_properties(cli_demo PROPERTIES PASS_REGULAR_EXPRESSION "evictions=1")
